bar_cat <- function(outcome, data, statistics = c("count"), # Can be "perc" 
                    title = NULL, horiz = F, fill = "pink") {

  dat <- calc_stats(
    data       = data,
    outcomes   = outcome,
    statistics = statistics
  ) 
  
  if(statistics == "perc"){dat$estimate <- dat$estimate * 100}

  x_lab <- dat$o_lab %>% unique()

  plt <- bar(
    dat       = dat,
    x         = o_cat,
    y         = estimate,
    x_lab     = x_lab,
    y_lab     = ifelse(statistics %in% "perc", "%", "Count"),
    title     = title, 
    fill_family  = fill,
    horiz = horiz
  )

  plt

}

bar_by <- function(
    data, 
    outcome, 
    predictor = NULL, 
    cluster = F,
    statistics = c("perc"), # Can be "count"
    title = NULL, 
    horiz = F, 
    palette = NULL,
    show_total = T, 
    x_lab = NULL
    ) {
  
  stopifnot(length(statistics) == 1)
  
  if(statistics %in% c("mean", "median")){
    chart_type = "num"} else {chart_type = "cat"}
  
  fill_arg <- "pink"
  
    dat <- 
      calc_stats(
        data       = data,
        outcomes   = outcome,
        predictors = predictor,
        statistics = statistics
        ) 
  
  if(chart_type == "num" & is.null(predictor)){
    
    chart_func <- bar
    x_arg <- sym("o_lab")
    x_lab_arg = x_lab

  } else if (chart_type == "cat" & is.null(predictor)){
    
    chart_func <- bar
    x_arg <- sym("o_cat")
    x_lab_arg <- dat$o_lab[1]
  
  } else if (chart_type == "num" & !is.null(predictor)){
    
    chart_func <- bar
  
  } else if (chart_type == "cat" & !is.null(predictor) & cluster == F){
    
    chart_func <- bar_stacked
    fill_arg <- sym("o_cat")

  } else {chart_func <- bar_clustered}
    
  if(!is.null(predictor)){
    x_arg <- sym("p_cat1")
    x_lab_arg <- dat$p_lab1 %>% grep("Total", ., value = T, invert = T) %>% unique()
    fill_arg <- sym("o_cat")
  }
    
  if(is.null(palette) & identical(chart_func, bar)){palette <- "pink"} 
  
  if(show_total==F & !is.null(predictor)){dat <- dat %>% filter(p_cat1 != "Total")}
  if(statistics == "perc"){dat$estimate <- dat$estimate * 100}

  rlang::inject(
    chart_func(dat = dat, 
                x = !!x_arg, 
                y = estimate,
                fill = !!fill_arg,
                fill_family = palette,
                x_lab = x_lab_arg,
                y_lab = ifelse(statistics %in% "perc", "%", "Count"),
                title = title,
                horiz = horiz
    )
  )
}
